// ============================================================================
// Bridge Node v1.0 — Token Sniffer + Distance Printer
//
// Bridge là một DWM1001 riêng (không làm ranging, không tham gia token ring).
// Nó chỉ lắng nghe (RX liên tục) toàn bộ token frame do các tag phát.
// Mỗi khi nhận được token frame hợp lệ, nó decode distances và in CSV ra UART.
//
// Tại sao không collision:
//   - Bridge KHÔNG bao giờ TX → không chiếm channel.
//   - Token frame đã được TX bởi tag khi nó KHÔNG đang ranging
//     (ranging và token_send() là tuần tự, không overlap).
//   - Bridge chỉ parse token frame, không cần timing chính xác.
//
// Token frame layout (phải khớp với ss_initiator_tag.c):
//   Byte  0– 9: IEEE 802.15.4a MAC header
//   Byte 10–11: TOKEN_FROM  (uint16 little-endian)
//   Byte 12–13: TOKEN_TO    (uint16 little-endian)
//   Byte 14–15: cycle count (uint16 little-endian)
//   Byte 16+  : NUM_ANCHORS × 6 bytes
//                 [0–1] anchor_id (uint16 LE)
//                 [2–5] distance  (int32 LE, unit = 0.1 mm; 0xFFFFFFFF = invalid)
//
// Output CSV (UART):
//   TagID,Cycle,AnchorID,Distance_mm,Valid
//   0x5678,1,0x1001,1234.5,1
//   0x5678,1,0x1002,985.0,1
//   ...
//
// PHY config: phải HOÀN TOÀN khớp với tag (channel, PRF, preamble, PAC, SFD, rate)
//   .txPreambLength = DWT_PLEN_128
//   .rxPAC          = DWT_PAC8
//   .sfdTO          = 129
// ============================================================================

#include <stdio.h>
#include <string.h>
#include "FreeRTOS.h"
#include "task.h"
#include "nrf.h"
#include "deca_device_api.h"
#include "deca_regs.h"
#include "port_platform.h"

// ============================================================================
// CONFIG — phải khớp với ss_initiator_tag.c
// ============================================================================
#define NUM_TAGS    2
#define NUM_ANCHORS 3

// Phải khớp với TAG_IDS[] trong tag firmware để validate from-field
static const uint16_t TAG_IDS[]    = { 0x5678, 0x5679, 0x567A, 0x567B, 0x567C, 0x567D };
static const uint32_t ANCHOR_IDS[] = { 0x1001, 0x1002, 0x1003, 0x1004 };

// ============================================================================
// TOKEN FRAME CONSTANTS — phải khớp với tag
// ============================================================================
#define ALL_MSG_SN_IDX          2
#define ALL_MSG_COMMON_LEN     10
#define TOKEN_FROM_IDX         10
#define TOKEN_TO_IDX           12
#define TOKEN_CYCLE_IDX        14
#define TOKEN_DATA_IDX         16
#define TOKEN_ANCHOR_STRIDE     6
#define TOKEN_HDR_LEN          16
#define TOKEN_MSG_LEN          (TOKEN_HDR_LEN + NUM_ANCHORS * TOKEN_ANCHOR_STRIDE)

// RX buffer: phải đủ lớn cho token frame
#define RX_BUF_LEN             80

// Header magic để nhận dạng token frame
static const uint8_t token_hdr_ref[] = {
    0x41, 0x88, 0, 0xCA, 0xDE, 'T', 'O', 'K', 'N', 0xE2
};

// ============================================================================
// BUFFERS
// ============================================================================
static uint8_t  rx_buffer[RX_BUF_LEN];
static uint32_t status_reg = 0;

// Stats
static uint32_t total_frames_rx   = 0;
static uint32_t valid_token_rx    = 0;
static uint32_t invalid_frames_rx = 0;
static uint8_t  header_printed    = 0;

// ============================================================================
// HELPERS
// ============================================================================
static uint16_t decode_u16_le(const uint8_t *p)
{
    return (uint16_t)p[0] | ((uint16_t)p[1] << 8);
}

static int32_t decode_i32_le(const uint8_t *p)
{
    return (int32_t)((uint32_t)p[0]
                   | ((uint32_t)p[1] << 8)
                   | ((uint32_t)p[2] << 16)
                   | ((uint32_t)p[3] << 24));
}

// Kiểm tra from_id có hợp lệ không (là 1 trong NUM_TAGS tag ID đã biết)
static int is_known_tag(uint16_t id)
{
    for (int i = 0; i < NUM_TAGS; i++)
        if (TAG_IDS[i] == id) return 1;
    return 0;
}

// ============================================================================
// PARSE & PRINT một token frame
// ============================================================================
static void process_token_frame(const uint8_t *buf, uint32_t flen)
{
    // Validate minimum length
    if (flen < (uint32_t)TOKEN_MSG_LEN) {
        invalid_frames_rx++;
        return;
    }

    uint16_t from_id  = decode_u16_le(&buf[TOKEN_FROM_IDX]);
    uint16_t to_id    = decode_u16_le(&buf[TOKEN_TO_IDX]);
    uint16_t cycle    = decode_u16_le(&buf[TOKEN_CYCLE_IDX]);

    if (!is_known_tag(from_id)) {
        invalid_frames_rx++;
        return;
    }

    valid_token_rx++;

    // Print CSV header once:
    // TagID,Cycle,D_1001,D_1002,D_1003,D_1004   (-1 = invalid/no response)
    if (!header_printed) {
        printf("TagID,Cycle");
        for (int i = 0; i < NUM_ANCHORS; i++)
            printf(",D_%04X", (unsigned)ANCHOR_IDS[i]);
        printf("\r\n");
        header_printed = 1;
    }

    // TagID + Cycle
    printf("0x%04X,%u", from_id, cycle);

    // Tất cả anchor distance trên cùng 1 dòng
    for (int i = 0; i < NUM_ANCHORS; i++) {
        const uint8_t *p = &buf[TOKEN_DATA_IDX + i * TOKEN_ANCHOR_STRIDE];
        int32_t d_raw = decode_i32_le(p + 2);
        if (d_raw == (int32_t)0xFFFFFFFF)
            printf(",-1");               // anchor không trả lời
        else
            printf(",%.0f", d_raw / 10.0);  // 0.1mm unit → mm
    }

    printf("\r\n");
}

// ============================================================================
// BRIDGE MAIN LOOP
// Bridge ở chế độ RX liên tục, không TX.
// ============================================================================
static void bridge_rx_loop(void)
{
    // Enable RX
    dwt_forcetrxoff();
    dwt_write32bitreg(SYS_STATUS_ID, 0xFFFFFFFF);
    dwt_rxenable(DWT_START_RX_IMMEDIATE);

    while (1) {
        status_reg = dwt_read32bitreg(SYS_STATUS_ID);

        // --- Frame nhận thành công ---
        if (status_reg & SYS_STATUS_RXFCG) {
            dwt_write32bitreg(SYS_STATUS_ID, SYS_STATUS_RXFCG);

            uint32_t flen = dwt_read32bitreg(RX_FINFO_ID) & RX_FINFO_RXFLEN_MASK;
            if (flen <= RX_BUF_LEN) {
                memset(rx_buffer, 0, RX_BUF_LEN);
                dwt_readrxdata(rx_buffer, flen, 0);
                //printf("RX RAW (%lu bytes): ", flen);
                //for (uint32_t i = 0; i < flen; i++) {
                //    printf("%02X ", rx_buffer[i]);
                //}
                //printf("\r\n");
            } else {
                // Frame quá lớn, bỏ qua
                dwt_rxenable(DWT_START_RX_IMMEDIATE);
                invalid_frames_rx++;
                continue;
            }

            total_frames_rx++;

            // Bỏ qua sequence number khi so sánh header
            uint8_t saved_sn = rx_buffer[ALL_MSG_SN_IDX];
            rx_buffer[ALL_MSG_SN_IDX] = 0;

            if (memcmp(rx_buffer, token_hdr_ref, ALL_MSG_COMMON_LEN) == 0) {
                // Là token frame — parse và in
                process_token_frame(rx_buffer, flen);
            }
            // Ranging frames (poll/resp) bị bỏ qua yên lặng — bridge không quan tâm

            rx_buffer[ALL_MSG_SN_IDX] = saved_sn; // restore (không thực sự cần)
            dwt_rxenable(DWT_START_RX_IMMEDIATE);
        }

        // --- RX error hoặc timeout ---
        else if (status_reg & (SYS_STATUS_ALL_RX_TO | SYS_STATUS_ALL_RX_ERR)) {
            dwt_write32bitreg(SYS_STATUS_ID,
                              SYS_STATUS_ALL_RX_TO | SYS_STATUS_ALL_RX_ERR);
            dwt_rxreset();
            dwt_rxenable(DWT_START_RX_IMMEDIATE);
        }

        // Yield CPU để FreeRTOS không bị starvation
        vTaskDelay(0);
    }
}

// ============================================================================
// PERIODIC STATS (optional — in ra mỗi 10 giây)
// Có thể tắt nếu không cần
// ============================================================================
#define STATS_INTERVAL_MS 10000

static TickType_t last_stats_tick = 0;

static void maybe_print_stats(void)
{
    TickType_t now = xTaskGetTickCount();
    if (now - last_stats_tick >= pdMS_TO_TICKS(STATS_INTERVAL_MS)) {
        last_stats_tick = now;
        printf("# BRIDGE STATS: total_rx=%lu valid_token=%lu invalid=%lu\r\n",
               (unsigned long)total_frames_rx,
               (unsigned long)valid_token_rx,
               (unsigned long)invalid_frames_rx);
    }
}

// ============================================================================
// TASK ENTRY — Bridge không cần token, langsung RX
// ============================================================================
void ss_initiator_task_function(void *pvParameter)
{
    UNUSED_PARAMETER(pvParameter);

    dwt_setleds(DWT_LEDS_ENABLE);

    printf("\r\n========================================\r\n");
    printf("Bridge Node v1.0\r\n");
    printf("Mode: RX-only token sniffer\r\n");
    printf("Monitoring %d tags, %d anchors\r\n", NUM_TAGS, NUM_ANCHORS);
    printf("Token frame: %d bytes expected\r\n", TOKEN_MSG_LEN);
    printf("Output: CSV on UART\r\n");
    printf("========================================\r\n\r\n");

    last_stats_tick = xTaskGetTickCount();

    bridge_rx_loop();
}
